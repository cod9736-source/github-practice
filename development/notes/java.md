<summary>학생 입력삭제</summary>

do while
![](../assets/b6cd8de40840aef94f48.png)
![](../assets/affa1f1cb9200297c01f.png)
불리안
```text
package quiz;

import java.util.ArrayList;
import java.util.Scanner;

public class Student_Program {
    public static void main(String[] args) {
        // 1. 메뉴를 보여주고 사용자에게 메뉴 입력 받기 반복
        Scanner sc = new Scanner(System.in);
        ArrayList<String> students = new ArrayList<>();

        boolean check = true;

        while (check) {
            System.out.println("========== 학생관리 프로그램 ==========");
            System.out.println("1.학생 등록 2.학생 삭제 3.프로그램 종료");
            System.out.print("메뉴 입력: ");

            int opt = sc.nextInt();
            sc.nextLine();
            // nextInt(넥스트 인트) 다음에 남아있는 엔터를 제거하려고 사용

            String name;
            // name(네임) = 학생 이름 저장 변수
            // switch 안에서 여러 case가 같이 쓰니까 switch 전에 선언

            switch (opt) {

            case 1:
                System.out.print("학생 이름: ");
                name = sc.nextLine();

                students.add(name);
                System.out.println(name + " 학생이 등록되었습니다.");
                break;

            case 2:
                System.out.print("학생 이름: ");
                name = sc.nextLine();

                if (students.contains(name)) {
                    // contains(컨테인스) = 리스트 안에 해당 값이 있는지 확인
                    students.remove(name);
                    // remove(리무브) = 리스트에서 해당 값 삭제
                    System.out.println(name + " 학생이 삭제되었습니다.");
                } else {
                    System.out.println("없는 학생입니다. 다시 입력하세요.");
                }
                break;

            case 3:
                System.out.println("학생목록: " + students);
                System.out.printf("총 %d명의 학생이 등록되었습니다.\n", students.size());
                check = false;
                break;

            default:
                System.out.println("잘 못 입력 하셨습니다.");
                break;
            }
        }

        sc.close();
    }
}
```
